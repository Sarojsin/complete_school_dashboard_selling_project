```python
from typing import Optional

class ProgramBase(BaseModel):
    name: str
    code: str

class ProgramCreate(ProgramBase):
    pass

```

**repository.py:**
```python
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from typing import List, Optional
from .models import ProgramModel

class ProgramRepository:
    def __init__(self, db: AsyncSession):
        self.db = db

    async def get_all(self, skip: int = 0, limit: int = 100) -> List[ProgramModel]:
        result = await self.db.execute(select(ProgramModel).offset(skip).limit(limit))
        return list(result.scalars().all())

    async def get_by_id(self, program_id: int) -> Optional[ProgramModel]:
        result = await self.db.execute(select(ProgramModel).where(ProgramModel.id == program_id))
        return result.scalar_one_or_none()

__all__ = ["ProgramRepository"]
```

**service.py:**
```python
from typing import Optional, List
from .repository import ProgramRepository
from .schemas import ProgramResponse

class ProgramService:
    def __init__(self, repository: ProgramRepository):
        self.repository = repository

    async def get_all_programs(self, skip: int = 0, limit: int = 100) -> List[ProgramResponse]:
        programs = await self.repository.get_all(skip, limit)
        return [ProgramResponse.model_validate(p) for p in programs]

    async def get_program(self, program_id: int) -> Optional[ProgramResponse]:
        program = await self.repository.get_by_id(program_id)
        if program:
            return ProgramResponse.model_validate(program)
        return None

__all__ = ["ProgramService"]
```

**router.py:**
```python
from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.ext.asyncio import AsyncSession
from modules.college.database import get_college_async_db
from modules.auth.dependencies import get_current_user, require_college_portal
from modules.shared.models import User
from .repository import ProgramRepository
from .service import ProgramService

router = APIRouter(prefix="/programs", tags=["College Programs"], dependencies=[Depends(require_college_portal)])

@router.get("/")
async def list_programs(
    skip: int = 0,
    limit: int = 100,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_college_async_db)
):
    repo = ProgramRepository(db)
    service = ProgramService(repo)
    return await service.get_all_programs(skip, limit)

@router.get("/{program_id}")
async def get_program(
    program_id: int,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_college_async_db)
):
    repo = ProgramRepository(db)
    service = ProgramService(repo)
    program = await service.get_program(program_id)
    if not program:
        raise HTTPException(status_code=404, detail="Program not found")
    return program

__all__ = ["router"]
```

**Example transformation for college_hod:**

Current api.py:
```python
@router.get("/dashboard")
async def get_hod_dashboard(current_user: User = Depends(get_current_user), db: AsyncSession = Depends(...)):
    result = await db.execute(select(Department).where(Department.hod_teacher_id == current_user.id))

```

Refactored:
```python
# repository.py
class HodRepository:
    def __init__(self, db: AsyncSession):
        self.db = db
    async def get_departments_by_hod(self, hod_user_id: int):
        result = await self.db.execute(select(Department).where(Department.hod_teacher_id == hod_user_id))
        return result.scalars().all()

# service.py
class HodService:
    def __init__(self, repository: HodRepository):
        self.repository = repository
    async def get_dashboard(self, user_id: int):
        departments = await self.repository.get_departments_by_hod(user_id)
        return {
            "departments_count": len(departments),
            "departments": departments
        }

# router.py
@router.get("/dashboard")
async def get_hod_dashboard(current_user: User = Depends(get_current_user), db: AsyncSession = Depends(...)):
```
